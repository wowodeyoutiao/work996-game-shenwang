[@pageup]
#IF
Large N$n_page 1
#Act
Dec N$n_page 1
Goto @首爆界面UI
Break
#ElseAct
Mov N$n_page <$STR(N$b_page)>

[@pagedown]
#IF

#Act
Inc N$n_page 1
Goto @首爆界面UI
Break
#ElseAct
Goto @Main


[@取首爆数据]
#IF
#Act
MOV S$首爆UI
MOV N$Y1 110
Mul n$now_line <$STR(N$n_page)> 10
FORMULATION <$STR(n$now_line)>-10 n$now_line_d

MOV N1 <$STR(n$now_line_d)>
while N1 < <$STR(N$now_line)>
MOV S$item_name <$cfg_首爆奖励配置(<$STR(N1)>_0)>
GETSTRVALUE A16 <$cfg_首爆奖励配置(<$STR(N1)>_0)> N$剩余名额

INC N1 1
endwhile
